    const host = (window as any).__cryptoidDeviceReplay;
    if (!host) return;
    const start = (mode: { enemies: number; double: boolean; shield: boolean }) => {
      clearControls();
      const { width, height } = fieldSizeRef.current;
      const s = createInitialState();
      Object.assign(s, { sector: 12, section: 12, chainBlocks: 1, status: 'playing', phase: 'ATTACK_CYCLE',

        shieldActive: mode.shield, shieldCharges: mode.shield ? 1 : 0, shieldMs: mode.shield ? 120000 : 0,
        powerUpTypes: ['shield','overdrive','weapon','rapid'] });
      const slots = createFormationSlots(12,12,width,height,visibleTopRef.current,playerDiameterRef.current);
      s.asteroids = Array.from({ length: mode.enemies },(_,i) => {
        const enemy = spawnAsteroid(nextIdRef.current++,width,visibleTopRef.current,i,12,slots);
        enemy.entryElapsed=enemy.entryDuration;enemy.formationElapsed=enemy.formationDuration;
        enemy.health=enemy.maxHealth=1000000;

      formationOffsetRef.current=0;flightRef.current=0;formationStartedRef.current=true;
      sectionSlotsRef.current=slots;sectionElapsedRef.current=20000;clearTimerRef.current=0;

      varietyRef.current=createCombatVariety(true);lastPlayerRef.current={...s.player};

      return { sector:s.sector,phase:s.phase,status:s.status,enemies:s.asteroids.length,
        shots:s.shots.length,enemyShots:s.enemyShots.length,effects:s.effects.length,
        shield:s.shieldActive&&s.shieldMs>0,weaponLevel:s.weaponLevel,pickupWeaponMs:s.pickupWeaponMs,
        fieldWidth:fieldSizeRef.current.width,fieldHeight:fieldSizeRef.current.height,

      for(const enemy of s.asteroids)enemy.health=enemy.maxHealth=1000000;
      return false;
    };
    return host.install({start,read,sustain});
